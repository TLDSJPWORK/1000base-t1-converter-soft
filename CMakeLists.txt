cmake_minimum_required(VERSION 3.13)
option(CONVERTER_HOST_TESTS "Build hardware-independent tests only" OFF)
if(CONVERTER_HOST_TESTS)
    project(converter_tests C)
    enable_testing()
    add_executable(test_mdio tests/test_mdio.c src/mdio_frame.c src/mdio_mmd.c src/phy_config.c)
    target_include_directories(test_mdio PRIVATE include)
    target_compile_options(test_mdio PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME mdio_protocol COMMAND test_mdio)
    return()
endif()
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to an installed Raspberry Pi Pico SDK (tested: 2.2.0)")
endif()
set(PICO_BOARD converter_rp2040)
list(APPEND PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/include/boards")
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(converter_soft C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()

set(CONVERTER_T1_ROLE "diagnostic" CACHE STRING
    "T1 role applied at boot: diagnostic, master, or slave")
set_property(CACHE CONVERTER_T1_ROLE PROPERTY STRINGS diagnostic master slave)
if(CONVERTER_T1_ROLE STREQUAL "master")
    set(CONVERTER_T1_ROLE_DEFINE CONVERTER_T1_ROLE_MASTER=1)
    set(CONVERTER_OUTPUT_NAME converter_soft_master)
elseif(CONVERTER_T1_ROLE STREQUAL "slave")
    set(CONVERTER_T1_ROLE_DEFINE CONVERTER_T1_ROLE_SLAVE=1)
    set(CONVERTER_OUTPUT_NAME converter_soft_slave)
elseif(CONVERTER_T1_ROLE STREQUAL "diagnostic")
    set(CONVERTER_T1_ROLE_DEFINE CONVERTER_T1_ROLE_DIAGNOSTIC=1)
    set(CONVERTER_OUTPUT_NAME converter_soft)
else()
    message(FATAL_ERROR "CONVERTER_T1_ROLE must be diagnostic, master, or slave")
endif()

add_executable(converter_soft src/main.c src/mdio_frame.c src/mdio_pio.c src/mdio_mmd.c src/phy_config.c)
target_include_directories(converter_soft PRIVATE include)
target_compile_definitions(converter_soft PRIVATE ${CONVERTER_T1_ROLE_DEFINE})
target_compile_options(converter_soft PRIVATE -Wall -Wextra -Werror)
target_link_libraries(converter_soft pico_stdlib hardware_pio hardware_clocks)
pico_generate_pio_header(converter_soft "${CMAKE_CURRENT_LIST_DIR}/src/mdio.pio")
pico_enable_stdio_usb(converter_soft 1)
pico_enable_stdio_uart(converter_soft 0)
pico_set_program_name(converter_soft "1000base-t1-converter-soft")
pico_set_program_version(converter_soft "0.3.0")
set_target_properties(converter_soft PROPERTIES OUTPUT_NAME ${CONVERTER_OUTPUT_NAME})
pico_add_extra_outputs(converter_soft)
